namespace app\demo;

import "std";

/// A generic payload container.
object Payload::<T> {
    data: [T],

    static fun init(data: [T]) > Payload::<T> {
        return Payload::<T>{ data = data };
    }
}

protocol Nameable {
    mut fun rename(name: str) > void;
}

enum<str> Direction {
    north,
    south,
}

fun count::<T>(items: [T]) > int {
    return items.len();
}

test "buzz syntax" {
    final payload = Payload::<int>{ data = [1, 2, 3] };
    var total = 0;

    foreach (index, value in payload.data): values {
        total += index + value;
    }

    final greeting = "total: {total}";
    final lines = `hello
{greeting}`;
    final pattern = $"total: \d+";
    final bucket = match (total) {
        0..5 -> "small",
        6, 7, 8 -> "medium",
        else -> "large",
    };

    std\assert(count::<int>(payload.data) == 3, message: greeting);
    _ = Direction.north.value as? str;
    _ = bucket;
    _ = pattern.match(lines);
}
